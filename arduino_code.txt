#define BLYNK_PRINT Serial
#define BLYNK_TEMPLATE_ID "TMPL3N6kureIp"
#define BLYNK_TEMPLATE_NAME "TALKING PLANT"
#define BLYNK_AUTH_TOKEN "5fcwOuS2056RNIZvUN3F5JxmmsvfIDkt"

#include <ESP8266WiFi.h>
#include <BlynkSimpleEsp8266.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// WiFi credentials
char ssid[] = "PLANT";
char pass[] = "12345678";

// LCD
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Pin definitions
int v1 = D3;
int v2 = D4;
int relay = D5;
int soil = D0;

int flag = 0;
int flag1 = 0;

void setup()
{
  Serial.begin(9600);

  pinMode(v1, OUTPUT);
  pinMode(v2, OUTPUT);
  pinMode(relay, OUTPUT);
  pinMode(soil, INPUT);

  digitalWrite(relay, HIGH);
  digitalWrite(v1, HIGH);
  digitalWrite(v2, HIGH);

  Wire.begin(D2, D1); // SDA, SCL
  lcd.init();
  lcd.backlight();

  lcd.setCursor(0, 0);
  lcd.print(" PLANT WATERING ");
  lcd.setCursor(0, 1);
  lcd.print("   SYSTEM ");
  delay(3000);
  lcd.clear();

  Blynk.begin(BLYNK_AUTH_TOKEN, ssid, pass);
}

void loop()
{
  Blynk.run();

  int value = digitalRead(soil);
  Serial.println("Soil value: " + String(value));

  if (value == HIGH)
  {
    Serial.println("Soil is DRY");

    lcd.setCursor(0, 0);
    lcd.print("LAND IS DRY");
    lcd.setCursor(0, 1);
    lcd.print("WATERING ON");

    digitalWrite(relay, LOW); // Relay ON
    Blynk.virtualWrite(V0, "DRY");
    Blynk.virtualWrite(V1, "PUMP ON");

    if (flag == 0)
    {
      digitalWrite(v1, LOW);
      delay(500);
      digitalWrite(v1, HIGH);
      flag = 1;
      flag1 = 0;
    }
  }
  else
  {
    Serial.println("Soil is WET");

    lcd.setCursor(0, 0);
    lcd.print("LAND IS WET");
    lcd.setCursor(0, 1);
    lcd.print("PUMP OFF");

    digitalWrite(relay, HIGH); // Relay OFF
    Blynk.virtualWrite(V0, "WET");
    Blynk.virtualWrite(V1, "PUMP OFF");

    if (flag1 == 0)
    {
      digitalWrite(v2, LOW);
      delay(500);
      digitalWrite(v2, HIGH);
      flag1 = 1;
      flag = 0;
    }
  }

  delay(2000);
  lcd.clear();
}
